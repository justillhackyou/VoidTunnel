cmake_minimum_required(VERSION 3.21)
project(VoidTunnel VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
qt_add_executable(VoidTunnel WIN32 src/main.cpp)
target_link_libraries(VoidTunnel PRIVATE Qt6::Widgets Qt6::Network)
if(WIN32)
  target_link_libraries(VoidTunnel PRIVATE crypt32)
  target_compile_definitions(VoidTunnel PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
if(MSVC)
  target_compile_options(VoidTunnel PRIVATE /W4 /utf-8)
else()
  target_compile_options(VoidTunnel PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS VoidTunnel RUNTIME DESTINATION .)
enable_testing()
add_test(NAME self-test COMMAND VoidTunnel --self-test)
set_tests_properties(self-test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
